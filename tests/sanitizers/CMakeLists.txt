cmake_minimum_required(VERSION 3.21)
project(SlonyaraCoreSanitizers LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SLONYARA_CORE_SANITIZER "none" CACHE STRING "none, thread, or memory")
set_property(CACHE SLONYARA_CORE_SANITIZER PROPERTY STRINGS none thread memory)

if(NOT SLONYARA_CORE_SANITIZER STREQUAL "none")
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "Use Clang for TSan/MSan")
    endif()
    if(SLONYARA_CORE_SANITIZER STREQUAL "thread")
        add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -g)
        add_link_options(-fsanitize=thread)
    elseif(SLONYARA_CORE_SANITIZER STREQUAL "memory")
        if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
            message(FATAL_ERROR "The MSan suite requires Linux and an MSan-instrumented libc++")
        endif()
        add_compile_options(-fsanitize=memory -fsanitize-memory-track-origins=2
                            -fno-omit-frame-pointer -fPIE -g)
        add_link_options(-fsanitize=memory -fsanitize-memory-track-origins=2 -pie)
    else()
        message(FATAL_ERROR "Unknown SLONYARA_CORE_SANITIZER")
    endif()
endif()

if(SLONYARA_CORE_SANITIZER STREQUAL "memory")
    set(SLONYARA_MSAN_LIBCXX "" CACHE PATH "Prefix of an MSan-instrumented libc++")
    if(NOT EXISTS "${SLONYARA_MSAN_LIBCXX}/lib/libc++.a")
        message(FATAL_ERROR "Build instrumented libc++ with tools/build-msan-libcxx.sh first")
    endif()
    add_compile_options(-stdlib=libc++ -nostdinc++)
    include_directories(SYSTEM "${SLONYARA_MSAN_LIBCXX}/include/c++/v1")
    add_link_options(-stdlib=libc++ "-L${SLONYARA_MSAN_LIBCXX}/lib")
endif()

find_package(Threads REQUIRED)
add_executable(CoreSanitizerTest coretests.cpp)
target_include_directories(CoreSanitizerTest PRIVATE ../../src)
target_link_libraries(CoreSanitizerTest PRIVATE Threads::Threads)
if(SLONYARA_CORE_SANITIZER STREQUAL "memory")
    target_link_libraries(CoreSanitizerTest PRIVATE
        "-Wl,--start-group" c++ c++abi unwind "-Wl,--end-group")
endif()
target_compile_options(CoreSanitizerTest PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME CoreEncoding COMMAND CoreSanitizerTest encoding)
add_test(NAME CoreCancellation COMMAND CoreSanitizerTest cancellation)
set_tests_properties(CoreEncoding CoreCancellation PROPERTIES TIMEOUT 90)

if(NOT SLONYARA_CORE_SANITIZER STREQUAL "none")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(SanitizerProbe probe.cpp)
    target_link_libraries(SanitizerProbe PRIVATE Threads::Threads)
    if(SLONYARA_CORE_SANITIZER STREQUAL "memory")
        target_compile_definitions(SanitizerProbe PRIVATE SLONYARA_PROBE_MEMORY=1)
        target_link_libraries(SanitizerProbe PRIVATE
            "-Wl,--start-group" c++ c++abi unwind "-Wl,--end-group")
    endif()
    add_test(NAME SanitizerRuntime
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/verify-sanitizer.py
                ${SLONYARA_CORE_SANITIZER} $<TARGET_FILE:SanitizerProbe>)
    set_tests_properties(SanitizerRuntime PROPERTIES TIMEOUT 40)
endif()
